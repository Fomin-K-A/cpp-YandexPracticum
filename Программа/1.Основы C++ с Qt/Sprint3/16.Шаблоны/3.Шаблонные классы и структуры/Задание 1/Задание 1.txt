Vector2D<int> my_vector{{1, 2}, {4, 6}};

assert((my_vector.GetStart() == Point<int>{1, 2}));
assert((my_vector.GetEnd() == Point<int>{4, 6}));
assert((my_vector.GetLengthSq() == 25));

my_vector.SetStart({4, 6});

assert((my_vector.GetStart() == Point<int>{4, 6}));
assert((my_vector.GetEnd() == Point<int>{4, 6}));
assert((my_vector.GetLengthSq() == 0));

my_vector.SetEnd({1, 2});
  
assert((my_vector.GetStart() == Point<int>{4, 6}));
assert((my_vector.GetEnd() == Point<int>{1, 2}));
assert((my_vector.GetLengthSq() == 25));

// Определим умножение и вычитание для строк.
// Сложение для них уже определено.
namespace std {
inline std::string operator*(const std::string& l, const std::string& r) {
    return "(" + l + "*" + r + ")";
}
inline std::string operator-(const std::string& l, const std::string& r) {
    return l + "-" + r;
}
} 

Vector2D<std::string> my_vector{{"hello", "I"}, {"❤️", "C++"}}; 
assert((my_vector.GetStart() == Point<std::string>{"hello", "I"}));
assert((my_vector.GetEnd() == Point<std::string>{"❤️", "C++"}));

// Так как мы определили для std::string
// все используемые в шаблоне операции,
// можем посчитать "длину" вектора со строковыми координатами.
assert((my_vector.GetLengthSq() == "(hello-❤️*hello-❤️)(I-C++*I-C++)" ||
        my_vector.GetLengthSq() == "(❤️-hello*❤️-hello)(C++-I*C++-I)"));
