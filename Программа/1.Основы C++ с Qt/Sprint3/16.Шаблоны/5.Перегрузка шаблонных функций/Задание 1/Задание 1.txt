Представьте простую геометрическую задачу: расчёт периметра четырёхугольника. Для её решения можно использовать различные исходные данные: если четырёхугольник неправильный, необходимо знать все его стороны. Для прямоугольника или параллелограмма достаточно двух сторон. А чтобы найти периметр квадрата, хватит и одной стороны.
Что нужно сделать
Напишите перегруженную шаблонную функцию GetPerimeter, которая работает для всех трёх вариантов входных данных:
для одного аргумента — вычисляет периметр ромба: умножает переданное число на 4.
для двух аргументов — вычисляет периметр параллелограмма по формуле (a + b) * 2.
для четырёх аргументов — вычисляет периметр произвольного четырёхугольника по формуле a + b + c + d.
Использовать её можно так:

// Можно передать четыре стороны.
assert((GetPerimeter(4, 5, 6, 7) == 22));
// Или две.
assert((GetPerimeter(4, 5) == 18));
// Или одну. 
assert((GetPerimeter(5) == 20)); 

// Можно передать длины разных типов.
assert((GetPerimeter<int, double>(4, 1.1) == 10.2)); 
Функция должна быть шаблонной, то есть уметь работать не только с типом int.
Каждый аргумент должен иметь свой шаблонный тип. Тип возвращаемого значения выведет компилятор — оставьте auto.
Юнит-тесты

assert((GetPerimeter(5) == 20));
assert((GetPerimeter(20, 10) == 60));
assert((GetPerimeter(4, 5, 6, 7) == 22));

assert((GetPerimeter(5.5) == 22.));
assert((GetPerimeter(0.5, 0.05) == 1.1));

assert((GetPerimeter<string>("a", "b", "c", "d") == "abcd"));

assert((GetPerimeter<int>(11.1, 22.2) == 66.4));
assert((GetPerimeter<int, int>(11.1, 22.2) == 66));

assert((GetPerimeter(1.5, 2.5, 3.3, 4.4) == 11.7));
assert((GetPerimeter<int>(1.5, 2.5, 3.3, 4.4) == 11.2));
assert((GetPerimeter<int, int>(1.5, 2.5, 3.3, 4.4) == 10.7));
assert((GetPerimeter<int, int, int>(1.5, 2.5, 3.3, 4.4) == 10.4));
assert((GetPerimeter<int, int, int, int>(1.5, 2.5, 3.3, 4.4) == 10));
